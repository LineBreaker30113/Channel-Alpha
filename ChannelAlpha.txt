TL  								/******************************************************************************************
TL  								 * Simple Mouse Handling class for that for knowing lot's of things about mouse.
TL  								 * Stores position of the mouse in "currentMousePosition" where you call "lbUpdate()".
TL  								 * If you set the value of "mouseOriginObject" to a component, "position"
TL  								 * will be relative to that component (best practice is setting it to a JPanel).
TL  								 * For using this object you can either add this as a "MouseListener" to a "JPanel"
TL  								 * (not to a "JFrame") or you can call "onPress", "onRelease" on related functions
TL  								 * but for handling "mouseInside" you need to set it on your own,
TL  								 * but hopefully it is not changing any of other methods working style.
TL  								 * Boolean "left", "right", "middle" indicates if they are down.
TL  								 * Booleans that ends with "Readed" are set to false only when
TL  								 * the related key released automatically and there is not
TL  								 * any other action, so YOU can handle button presses.
TL  								 * The "wheel" starts at 0 and as wheel events happen it changes,
TL  								 * so you can learn scroll of wheel relative to the one at start of application.
TL  								 * Integers that end with "for" meant to tell you how many updates they stayed
TL  								 * in the same state, for using this utility you must call lbUpdate() in every update.
TL  								 ******************************************************************************************/
TL  								public class MouseUpdateTracker implements MouseListener, MouseWheelListener {
TL  									public long leftLastPressDate = 0, rightLastPressDate = 0, middleLastPress = 0;
TL  									public long leftLastReleaseDate = 0, rightLastReleaseDate = 0, middleLastRelease = 0;
TL  									public boolean left = false, right = false, midle = false, mouseInside = false;
TL  									public boolean leftReaded = false, rightReaded = false, middleReaded = false;
TL  									
TL  									public Component component;
TL  								
TL  									public MouseUpdateTracker(Component component) {
TL  										this.component = component;
TL  										leftLastPressDate = new Date().getTime(); rightLastPressDate = new Date().getTime();
TL  										middleLastPress = new Date().getTime(); middleLastRelease = new Date().getTime();
TL  										leftLastReleaseDate = new Date().getTime(); rightLastReleaseDate = new Date().getTime();
TL  									}
TL  								
TL  									public Point leftLastPressPosition = null, leftLastReleasePosition = null;
TL  									public Point rightLastPressPosition = null, rightLastReleasePosition = null;
TL  									public int leftSameFor = 0, rightSameFor = 0;
TL  									public double wheelStatus = 0;
TL  									public Component mouseOriginObject = null;
TL  									public Point position = new Point();
TL  									public void onPress(MouseEvent e) {
TL  										switch(e.getButton()) {
TL  										case MouseEvent.BUTTON2: midle = true; break;
TL  										case MouseEvent.BUTTON1:
TL  											left = true; leftSameFor = 0;
TL  											leftLastPressPosition = new Point(e.getX(), e.getY());
TL  											leftLastPressDate = new Date().getTime();
TL  											break;
TL  										case MouseEvent.BUTTON3:
TL  											right = true; rightSameFor = 0;
TL  											rightLastPressPosition = new Point(e.getX(), e.getY());
TL  											rightLastPressDate = new Date().getTime();
TL  											break;
TL  										}
TL  									}
TL  									public void onRelease(MouseEvent e) {
TL  										switch(e.getButton()) {
TL  										case MouseEvent.BUTTON2: midle = false; middleReaded = false; break;
TL  										case MouseEvent.BUTTON1:
TL  											left = false; leftSameFor = 0; leftReaded = false;
TL  											leftLastReleasePosition = new Point(e.getX(), e.getY());
TL  											leftLastReleaseDate = new Date().getTime();
TL  											break;
TL  										case MouseEvent.BUTTON3:
TL  											right = false; rightSameFor = 0; rightReaded = false;
TL  											rightLastReleasePosition = new Point(e.getX(), e.getY());
TL  											leftLastReleaseDate = new Date().getTime();
TL  											break;
TL  										}
TL  									}
TL  									public void lbUpdate() {
TL  										if(mouseOriginObject == null) { position = getPosition();
TL  										} else { position = getPosition(mouseOriginObject); }
TL  										leftSameFor++; rightSameFor++;
TL  									}
TL  									public Point getPosition() {
TL  										Point p = MouseInfo.getPointerInfo().getLocation();
TL  										return new Point(p.x, p.y);
TL  									}
TL  									public Point getRelativePosition() {
TL  										Point p = component.getLocationOnScreen(); Point r = getPosition();
TL  										r.x -= p.x; r.y -= p.y;
TL  										return r;
TL  									}
TL  									public Point getPosition(Component relative) {
TL  										Point p = relative.getLocationOnScreen(); Point r = getPosition();
TL  										r.x -= p.x; r.y -= p.y;
TL  										return r;
TL  									}
